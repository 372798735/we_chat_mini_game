# 番茄闹钟待办清单数据库设计文档

## 1. 数据库概述

### 1.1 数据库选择

- **数据库类型**: MySQL 8.0+
- **部署方式**: 本地部署，随应用安装包一起分发
- **字符集**: utf8mb4
- **存储引擎**: InnoDB

### 1.2 设计原则

- 遵循第三范式，减少数据冗余
- 使用合理的索引，提升查询效率
- 保留修改历史，支持数据追踪
- 考虑数据备份和恢复需求

### 2.3 任务表 (t_task)

```sql
CREATE TABLE t_task (
    id BIGINT PRIMARY KEY AUTO_INCREMENT COMMENT '任务ID',
    user_id BIGINT NOT NULL COMMENT '用户ID',
    category_id BIGINT COMMENT '分类ID',
    title VARCHAR(100) NOT NULL COMMENT '任务标题',
    description TEXT COMMENT '任务描述',
    estimated_duration INT NOT NULL COMMENT '预计时长(分钟)',
    actual_duration INT DEFAULT 0 COMMENT '实际用时(分钟)',
    priority ENUM('low', 'medium', 'high') DEFAULT 'medium' COMMENT '优先级',
    status ENUM('pending', 'in_progress', 'completed', 'paused', 'cancelled') DEFAULT 'pending' COMMENT '任务状态',
    due_date TIMESTAMP NULL COMMENT '截止时间',
    reminder_time TIMESTAMP NULL COMMENT '提醒时间',
    tags VARCHAR(255) COMMENT '标签(逗号分隔)',
    sort_order INT DEFAULT 0 COMMENT '排序序号',
    parent_task_id BIGINT NULL COMMENT '父任务ID',
    is_recurring BOOLEAN DEFAULT FALSE COMMENT '是否循环',
    recurrence_rule VARCHAR(100) COMMENT '循环规则',
    completion_rate DECIMAL(5,2) DEFAULT 0.00 COMMENT '完成率',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
    completed_at TIMESTAMP NULL COMMENT '完成时间',
    FOREIGN KEY (user_id) REFERENCES t_user(id) ON DELETE CASCADE,
    FOREIGN KEY (category_id) REFERENCES t_category(id) ON DELETE SET NULL,
    FOREIGN KEY (parent_task_id) REFERENCES t_task(id) ON DELETE CASCADE,
    INDEX idx_user_id (user_id),
    INDEX idx_category_id (category_id),
    INDEX idx_status (status),
    INDEX idx_priority (priority),
    INDEX idx_due_date (due_date),
    INDEX idx_created_at (created_at),
    INDEX idx_user_status (user_id, status)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='任务表';
```

## 4. 视图设计

### 4.1 任务统计视图

```sql
CREATE VIEW v_task_statistics AS
SELECT
    u.id as user_id,
    u.username,
    COUNT(t.id) as total_tasks,
    SUM(CASE WHEN t.status = 'completed' THEN 1 ELSE 0 END) as completed_tasks,
    SUM(CASE WHEN t.status = 'in_progress' THEN 1 ELSE 0 END) as in_progress_tasks,
    SUM(t.actual_duration) as total_focus_time,
    AVG(CASE WHEN t.status = 'completed' THEN t.actual_duration ELSE NULL END) as avg_completion_time,
    MAX(t.completed_at) as last_completion_time
FROM t_user u
LEFT JOIN t_task t ON u.id = t.user_id
GROUP BY u.id, u.username;
```

### 4.2 每日番茄钟统计视图

```sql
CREATE VIEW v_daily_pomodoro AS
SELECT
    user_id,
    DATE(started_at) as date,
    COUNT(*) as total_pomodoros,
    SUM(CASE WHEN type = 'work' THEN 1 ELSE 0 END) as work_pomodoros,
    SUM(CASE WHEN type = 'short_break' THEN 1 ELSE 0 END) as short_breaks,
    SUM(CASE WHEN type = 'long_break' THEN 1 ELSE 0 END) as long_breaks,
    SUM(actual_duration) as total_minutes,
    AVG(actual_duration) as avg_duration
FROM t_pomodoro
GROUP BY user_id, DATE(started_at);
```

## 5. 存储过程

### 5.1 更新每日统计

```sql
DELIMITER //
CREATE PROCEDURE UpdateDailyStatistics(IN p_user_id BIGINT, IN p_date DATE)
BEGIN
    DECLARE v_total_tasks INT DEFAULT 0;
    DECLARE v_completed_tasks INT DEFAULT 0;
    DECLARE v_total_pomodoros INT DEFAULT 0;
    DECLARE v_total_focus_time INT DEFAULT 0;

    -- 统计任务数据
    SELECT COUNT(*), SUM(CASE WHEN status = 'completed' THEN 1 ELSE 0 END)
    INTO v_total_tasks, v_completed_tasks
    FROM t_task
    WHERE user_id = p_user_id AND DATE(created_at) = p_date;

    -- 统计番茄钟数据
    SELECT COUNT(*), SUM(actual_duration)
    INTO v_total_pomodoros, v_total_focus_time
    FROM t_pomodoro
    WHERE user_id = p_user_id AND DATE(started_at) = p_date;

    -- 更新或插入统计数据
    INSERT INTO t_statistics (
        user_id, stat_date, total_tasks, completed_tasks,
        total_pomodoros, total_focus_time, completion_rate
    ) VALUES (
        p_user_id, p_date, v_total_tasks, v_completed_tasks,
        v_total_pomodoros, v_total_focus_time,
        CASE WHEN v_total_tasks > 0 THEN (v_completed_tasks * 100.0 / v_total_tasks) ELSE 0 END
    )
    ON DUPLICATE KEY UPDATE
        total_tasks = v_total_tasks,
        completed_tasks = v_completed_tasks,
        total_pomodoros = v_total_pomodoros,
        total_focus_time = v_total_focus_time,
        completion_rate = CASE WHEN v_total_tasks > 0 THEN (v_completed_tasks * 100.0 / v_total_tasks) ELSE 0 END,
        updated_at = CURRENT_TIMESTAMP;
END //
DELIMITER ;
```

## 6. 触发器

### 6.1 任务完成触发器

```sql
DELIMITER //
CREATE TRIGGER tr_task_completed
AFTER UPDATE ON t_task
FOR EACH ROW
BEGIN
    IF OLD.status != 'completed' AND NEW.status = 'completed' THEN
        -- 更新实际用时
        IF NEW.actual_duration = 0 THEN
            UPDATE t_task
            SET actual_duration = TIMESTAMPDIFF(MINUTE, created_at, NEW.completed_at)
            WHERE id = NEW.id;
        END IF;

        -- 更新统计
        CALL UpdateDailyStatistics(NEW.user_id, DATE(NEW.completed_at));
    END IF;
END //
DELIMITER ;
```

### 6.2 番茄钟完成触发器

```sql
DELIMITER //
CREATE TRIGGER tr_pomodoro_completed
AFTER INSERT ON t_pomodoro
FOR EACH ROW
BEGIN
    -- 更新任务的实际用时
    UPDATE t_task
    SET actual_duration = actual_duration + NEW.actual_duration
    WHERE id = NEW.task_id;

    -- 更新统计
    CALL UpdateDailyStatistics(NEW.user_id, DATE(NEW.started_at));
END //
DELIMITER ;
```

### 7.2 备份脚本示例

```bash
#!/bin/bash
# 数据库备份脚本
DB_NAME="tomato_todo"
DB_USER="root"
DB_PASS="root"
BACKUP_DIR="/backup/mysql"
DATE=$(date +%Y%m%d_%H%M%S)

# 创建备份目录
mkdir -p $BACKUP_DIR

# 执行备份
mysqldump -u$DB_USER -p$DB_PASS --single-transaction --routines --triggers $DB_NAME > $BACKUP_DIR/backup_$DATE.sql

# 压缩备份文件
gzip $BACKUP_DIR/backup_$DATE.sql

# 记录备份信息
mysql -u$DB_USER -p$DB_PASS -e "
INSERT INTO $DB_NAME.t_backup_record (backup_type, backup_path, backup_size, status)
VALUES ('full', '$BACKUP_DIR/backup_$DATE.sql.gz', $(stat -c%s $BACKUP_DIR/backup_$DATE.sql.gz), 'success');
"

# 删除7天前的备份
find $BACKUP_DIR -name "backup_*.sql.gz" -mtime +7 -delete
```
